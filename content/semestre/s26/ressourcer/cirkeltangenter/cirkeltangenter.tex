\subsection{}
For hver af de følgende cirkler skal du
\begin{itemize}
  \item vise, at punktet $P$ ligger på cirklen,
  \item bestemme en ligning for tangenten i punktet $P$ (\href{https://plusbhf.systime.dk/?id=1458#c15432}{Se eksempel 4 i bogen})
\end{itemize}

\begin{enumerate}
  \item $\left(x-1\right)^2 + \left(y-2\right)^2 = 5$, \quad $P(3,3)$.
  \item $\left(x+4\right)^2 + \left(y-1\right)^2 = 10$, \quad $P(-7,2)$.
  \item $\left(x-2\right)^2 + \left(y+3\right)^2 = 20$, \quad $P(6,-5)$.
  \item $\left(x+1\right)^2 + \left(y+4\right)^2 = 5$, \quad $P(-3,-3)$.
  \item $x^2-4x+y^2+2y=0$, \quad $P(4,0)$.
  \item $x^2+6x+y^2-4y+8=0$, \quad $P(-1,3)$.
\end{enumerate}

\subsection{}
Afgør ved beregning, om linjen er tangent til cirklen (\href{https://plusbhf.systime.dk/?id=1458#c15431}{Se eksempel 5 i bogen})
\begin{enumerate}
  \item $y=3x-1$ og $\left(x-1\right)^2+\left(y-2\right)^2=10$.
  \item $y=-2x+7$ og $\left(x-3\right)^2+\left(y-1\right)^2=5$.
  \item $y=\frac{1}{2}x+2$ og $\left(x+2\right)^2+\left(y-1\right)^2=5$.
  \item $y=-\frac{3}{4}x+4$ og $\left(x-2\right)^2+\left(y+1\right)^2=9$.
\end{enumerate}

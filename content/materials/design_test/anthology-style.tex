% Classic novel: centred, quietly formal, with indented paragraphs.
\ifdefstring{\DesignFamily}{classic}{%
  \renewcommand{\editionopening}{\centering}
  \renewcommand{\editiontitlefont}{\normalfont\fontsize{28}{32}\selectfont}
  \renewcommand{\editionrule}{\par\vspace{6mm}{\color{hgblue}\rule{18mm}{.5pt}}\par\vspace{8mm}}
}{}
% Short fiction: intimate, unboxed, with a small italic byline.
\ifdefstring{\DesignFamily}{story}{%
  \renewcommand{\editionopening}{\centering}
  \renewcommand{\editiontitlefont}{\normalfont\fontsize{29}{33}\selectfont}
  \renewcommand{\editionrule}{\par\vspace{11mm}}
}{}
% Modernist novel: asymmetric opening and a more architectural title.
\ifdefstring{\DesignFamily}{modernist}{%
  \renewcommand{\editiontitlefont}{\plantinb\fontsize{36}{36}\selectfont}
  \renewcommand{\editionrule}{\par\vspace{5mm}{\color{hgblue}\hrule height 1.2pt}\vspace{7mm}}
}{}
% Personal essay: a literary-magazine opening and spacious block paragraphs.
\ifdefstring{\DesignFamily}{essay}{%
  \setlength{\parindent}{0pt}\setlength{\parskip}{.55\baselineskip}
  \renewcommand{\editiontitlefont}{\normalfont\itshape\fontsize{31}{34}\selectfont}
}{}
% Web writing: sans-serif reading texture, visible original subheadings.
\ifdefstring{\DesignFamily}{digital}{%
  \colorlet{editionaccent}{digitalteal}
  \setlength{\parindent}{0pt}\setlength{\parskip}{.65\baselineskip}
  \renewcommand{\editionbody}{\editorialsans\fontsize{10.5}{15}\selectfont}
  \renewcommand{\editiontitlefont}{\editorialsans\bfseries\fontsize{29}{32}\selectfont}
}{}
% Reportage: a larger headline and a strong burgundy editorial rule.
\ifdefstring{\DesignFamily}{reportage}{%
  \colorlet{editionaccent}{hgred}
  \setlength{\parindent}{0pt}\setlength{\parskip}{.4\baselineskip}
  \renewcommand{\editiontitlefont}{\plantinb\fontsize{33}{34}\selectfont}
  \renewcommand{\editionrule}{\par\vspace{4mm}{\color{hgred}\hrule height 2pt}\vspace{6mm}}
}{}
% Speech: formal metadata and more breathing room between spoken units.
\ifdefstring{\DesignFamily}{speech}{%
  \setlength{\parindent}{0pt}\setlength{\parskip}{.65\baselineskip}
  \renewcommand{\editionbody}{\fontsize{11.5}{16}\selectfont}
  \renewcommand{\editiontitlefont}{\plantin\fontsize{25}{28}\selectfont}
}{}
% Newspaper personal commentary: assertive headline, compact byline.
\ifdefstring{\DesignFamily}{commentary}{%
  \setlength{\parindent}{0pt}\setlength{\parskip}{.45\baselineskip}
  \renewcommand{\editiontitlefont}{\plantinb\fontsize{35}{37}\selectfont}
  \renewcommand{\editionrule}{\par\vspace{4mm}{\color{hgblue}\hrule height .4pt}\vspace{1mm}{\color{hgblue}\hrule height .4pt}\vspace{7mm}}
}{}
% Poetry: a smaller measure, preserved verse lines, and ample white space.
\ifdefstring{\DesignFamily}{poetry}{%
  \geometry{left=44mm,right=36mm}

  \renewcommand{\editionbody}{\fontsize{12.5}{19}\selectfont}
  \renewcommand{\editiontitlefont}{\normalfont\fontsize{29}{33}\selectfont}
  \renewcommand{\editionrule}{\par\vspace{13mm}}
}{}

% Print double-sided with "flip on long edge". Back pages are horizontally reversed
% so each word aligns with itself after cutting.
\newlength{\cardw}
\newlength{\cardh}
\setlength{\cardw}{\dimexpr(\paperwidth-24mm)/5\relax}
\setlength{\cardh}{\dimexpr(\paperheight-24mm)/5\relax}

\newcommand{\cardpage}[1]{%
\begin{tikzpicture}[remember picture,overlay]
  \coordinate (origin) at ([xshift=12mm,yshift=-12mm]current page.north west);
  \foreach \c in {0,...,5} {\draw[gray!65,line width=0.35pt] ([xshift=\c\cardw]origin) -- ([xshift=\c\cardw,yshift=-5\cardh]origin);}
  \foreach \r in {0,...,5} {\draw[gray!65,line width=0.35pt] ([yshift=-\r\cardh]origin) -- ([xshift=5\cardw,yshift=-\r\cardh]origin);}
  \foreach \word [count=\i from 0] in {#1} {%
    \pgfmathtruncatemacro{\col}{mod(\i,5)}%
    \pgfmathtruncatemacro{\row}{floor(\i/5)}%
    \node[align=center,text width=.88\cardw,font=\bfseries\fontsize{14}{16}\selectfont]
      at ([xshift=(\col+.5)*\cardw,yshift=-(\row+.5)*\cardh]origin) {\word};
  }
\end{tikzpicture}\newpage}

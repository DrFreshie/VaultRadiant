%% ─── Forløb environment ──────────────────────────────────────────────────────

\newcounter{ubcnt}
\gdef\ubCurTitel{}\gdef\ubCurDato{}\gdef\ubCurOmfang{}
\gdef\ubCurBog{}\gdef\ubCurKap{}\gdef\ubCurArb{}\gdef\ubCurFok{}

\newenvironment{forloeb}[4]{% {titel}{dato}{omfang}{bog}
  \stepcounter{ubcnt}%
  \gdef\ubCurTitel{#1}\gdef\ubCurDato{#2}%
  \gdef\ubCurOmfang{#3}\gdef\ubCurBog{#4}%
  \gdef\ubCurKap{}\gdef\ubCurArb{}\gdef\ubCurFok{}%
  \ubwriterow{\theubcnt}{#2}{#1}%
}{%
  \vspace{1em}%
  {\large\bfseries\overp\color{frbgreen}%
    Forløb \theubcnt: \ubCurTitel\par}%
  \vspace{0.4em}%
  \begin{tabularx}{\linewidth}{|l|X|}
    \hline
    Omfang: & \ubCurOmfang\ lektioner \\\hline
    Start:  & \ubCurDato \\\hline
    Indhold: &
      \ifdefempty{\ubCurBog}{}{%
        \textit{\ubCurBog}%
        \ifdefempty{\ubCurKap}{}{\newline\newline}%
      }%
      \ubCurKap
      \\\hline
    Arbejdsformer: & \ubCurArb \\\hline
    Fokuspunkter:  & \ubCurFok \\\hline
  \end{tabularx}%
}

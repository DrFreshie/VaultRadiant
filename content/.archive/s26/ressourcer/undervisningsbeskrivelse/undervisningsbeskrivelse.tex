%% ─── Overview table ──────────────────────────────────────────────────────────
%%
%% Rows are written to .aux on each compile and read back on the next.
%% The count (\ubfinalcnt) works the same way.
%% Result: correct overview from the 2nd lualatex run onward.

\newcommand{\ub@forlrows}{}
\newcommand{\ub@forlrow}[3]{\gappto\ub@forlrows{#1 & #2 & #3 \\\hline}}

\gdef\ubfinalcnt{?}
\AtEndDocument{%
  \immediate\write\@auxout{\string\gdef\string\ubfinalcnt{\theubcnt}}%
}

\newcommand{\forloebsoversigt}{%
  \vspace{1.5em}%
  {\large\bfseries\overp\color{frbgreen}Forløbsoversigt (\ubfinalcnt):\par}%
  \vspace{0.4em}%
  \begin{tabularx}{\linewidth}{|l|l|X|}
    \hline
    \ub@forlrows
  \end{tabularx}%
}

%% ─── Forløb environment ──────────────────────────────────────────────────────

\newcounter{ubcnt}

\newenvironment{forloeb}[4]{% {titel}{dato}{omfang}{bog}
  %% begin: reset accumulators, step counter, register for overview
  \stepcounter{ubcnt}%
  \gdef\ub@cur@titel{#1}%
  \gdef\ub@cur@dato{#2}%
  \gdef\ub@cur@omfang{#3}%
  \gdef\ub@cur@bog{#4}%
  \gdef\ub@cur@kap{}%
  \gdef\ub@cur@arb{}%
  \gdef\ub@cur@fok{}%
  \immediate\write\@auxout{\string\ub@forlrow{\theubcnt}{#2}{#1}}%
}{%
  %% end: render the detail table
  \vspace{1em}%
  {\large\bfseries\overp\color{frbgreen}%
    Forløb \theubcnt: \ub@cur@titel\par}%
  \vspace{0.4em}%
  \begin{tabularx}{\linewidth}{|l|X|}
    \hline
    Omfang: & \ub@cur@omfang\ lektioner \\\hline
    Start:  & \ub@cur@dato \\\hline
    Indhold: &
      \ifdefempty{\ub@cur@bog}{}{%
        \textit{\ub@cur@bog}%
        \ifdefempty{\ub@cur@kap}{}{\newline\newline}%
      }%
      \ub@cur@kap
      \\\hline
    Arbejdsformer: & \ub@cur@arb \\\hline
    Fokuspunkter:  & \ub@cur@fok \\\hline
  \end{tabularx}%
}

%% Bullet-item accumulators used inside {forloeb}
\newcommand{\kap}[1]{\gappto\ub@cur@kap{\textbullet\ #1\newline}}
\newcommand{\arb}[1]{\gappto\ub@cur@arb{\textbullet\ #1\newline}}
\newcommand{\fok}[1]{\gappto\ub@cur@fok{\textbullet\ #1\newline}}
